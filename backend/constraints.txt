#   pip install -c backend/constraints.txt ...
# so SAM-Audio's unpinned dependencies can't silently swap them out.

# PyPI, and a torchcodec that doesn't match torch ("Could not load libtorchcodec").

torchaudio==2.9.0
torchcodec==0.9.1
xformers==0.0.33.post1
transformers>=4.54.0,<5
